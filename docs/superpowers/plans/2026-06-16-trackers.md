> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build first-class plot trackers with persistence, base-context inclusion, chat-tool mutators, and player-sidebar display.

**Architecture:** Add a `Tracker` domain model with static registry semantics matching `ScheduledEvent` and `MysteryThread`. Wire the model through save/load, base-context preparation, mutator tool execution, `/api/player` serialization, and the existing chat-sidebar refresh path. Keep UI state client-side for hidden tracker visibility, while all tracker records remain persisted server-side.

**Tech Stack:** Node.js CommonJS, `node:test`, Nunjucks, Express route helpers in `api.js`/`server.js`, inline chat-page JavaScript in `views/index.njk`, SCSS compiled with `npm run scss:build:main`.

```js
const IdGenerator = require('./IdGenerator.js');

const VALID_TYPES = new Set(['countdown', 'numerical_count', 'x_out_of_total', 'percentage', 'short_string']);

function normalizeText(value) {
  return typeof value === 'string' ? value.trim() : '';
}

function normalizeBoolean(value) {
  return value === true;
}

function normalizeWorldMinute(value, label = 'lastUpdatedWorldMinute') {
  if (!Number.isInteger(value) || value < 0) {
    throw new Error(`Tracker ${label} must be a non-negative integer.`);
  }
  return value;
}

function normalizeType(value) {
  const type = normalizeText(value);
  if (!VALID_TYPES.has(type)) {
    throw new Error(`Tracker type must be one of: ${Array.from(VALID_TYPES).join(', ')}.`);
  }
  return type;
}

function validateValueForType(value, type) {
  const text = normalizeText(value);
  if (!text) {
    throw new Error('Tracker value must be a non-empty string.');
  }
  if (type === 'short_string') {
    const words = text.split(/\s+/).filter(Boolean);
    if (words.length > 3) {
      throw new Error('Tracker short_string value must be three words or fewer.');
    }
  }
  if (type === 'x_out_of_total' && !/^\d+\s*\/\s*\d+$/.test(text)) {
    throw new Error('Tracker x_out_of_total value must use x/total format.');
  }
  if (type === 'percentage' && !/^[+-]?(?:\d+(?:\.\d+)?|\.\d+)%$/.test(text)) {
    throw new Error('Tracker percentage value must be a finite number followed by %.');
  }
  return text;
}
```
